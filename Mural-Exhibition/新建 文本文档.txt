// 引入必需的 Node.js 模块
const express = require('express');
const fs = require('fs');
const path = require('path');

// 初始化 Express 应用
const app = express();
// 设置本地服务器的端口号
const PORT = 3000;

// 【关键步骤】告诉 Express 将 'public' 文件夹作为静态资源目录
// 这样我们在浏览器里就可以直接通过路径访问里面的 HTML 和 图片了
app.use(express.static('public'));

// 【核心逻辑】创建一个 API 接口供前端调用
app.get('/api/scan-assets', (req, res) => {
    // 定义 assets 文件夹的完整路径
    const assetsDirectory = path.join(__dirname, 'public', 'assets');
    const seriesList = [];

    try {
        // 1. 读取 assets 目录下的所有文件和文件夹名称
        const items = fs.readdirSync(assetsDirectory);

        // 2. 遍历每一个项目
        items.forEach(item => {
            const itemPath = path.join(assetsDirectory, item);
            
            // 3. 判断当前项目是不是一个“文件夹”
            if (fs.statSync(itemPath).isDirectory()) {
                // 如果是文件夹，说明这是一个“壁画系列”
                // 将相关信息打包成 JSON 格式
                seriesList.push({
                    folderName: item, // 文件夹名称，例如 "01_上层猖兵"
                    orgPath: `/assets/${item}/org.png`,     // 原图路径
                    linePath: `/assets/${item}/line.png`,   // 线稿路径
                    colorPath: `/assets/${item}/color.png`, // 上色路径
                    infoPath: `/assets/${item}/info.txt`    // 文本路径
                });
            }
        });

        // 4. 将扫描成功的数据发送给前端
        res.json({
            success: true,
            data: seriesList
        });

    } catch (error) {
        // 如果出错（例如还没创建 assets 文件夹），返回错误信息
        console.error("扫描文件夹失败:", error);
        res.status(500).json({
            success: false,
            message: "读取文件夹失败，请检查 public/assets 目录是否存在。"
        });
    }
});

// 启动服务器
app.listen(PORT, () => {
    console.log('=================================');
    console.log(`后台服务已成功启动！`);
    console.log(`请打开浏览器并访问: http://localhost:${PORT}`);
    console.log('按 Ctrl+C 可以关闭服务');
    console.log('=================================');
