        const webglContainer = document.getElementById('webgl-container');
        let scene, camera, renderer, lineMesh, colorPoints, shaderMaterial;
        let defaultCameraZ = 1.8; 
        let cameraOffsetX = 0;    
        
        const vertexShader = `
            uniform vec2 uMouse;
            uniform float uDispersion;
            uniform float uTime;
            varying vec2 vUv;
            void main() {
                vUv = uv;
                vec3 pos = position;
                float dist = distance(vUv, uMouse);
                float localDisp = 1.0 - smoothstep(0.05, 0.15, dist);
                float totalDisp = clamp(localDisp + uDispersion, 0.0, 1.0);
                
                float scatterX = sin(position.y * 15.0 + uTime * 2.0) * 0.1;
                float scatterY = cos(position.x * 15.0 + uTime * 2.0) * 0.1;
                float scatterZ = (sin(position.x * 10.0) + cos(position.y * 10.0) - 2.0) * 0.5;

                pos.x += scatterX * totalDisp; pos.y += scatterY * totalDisp; pos.z += scatterZ * totalDisp; 
                gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
                gl_PointSize = 2.5; 
            }
        `;

        const fragmentShader = `
            uniform sampler2D uColorTex;
            uniform vec2 uMouse;
            uniform float uDispersion;
            varying vec2 vUv;
            void main() {
                vec4 color = texture2D(uColorTex, vUv);
                if (color.a < 0.1) discard; 
                float dist = distance(vUv, uMouse);
                float localDisp = 1.0 - smoothstep(0.07, 0.15, dist);
                float totalDisp = clamp(localDisp + uDispersion, 0.0, 1.0);
                gl_FragColor = vec4(color.rgb, color.a * (1.0 - totalDisp));
            }
        `;

        function initWebGL() {
            scene = new THREE.Scene();
            const w = webglContainer.clientWidth || window.innerWidth;
            const h = webglContainer.clientHeight || window.innerHeight;
            
            camera = new THREE.PerspectiveCamera(45, w / h, 0.1, 100);
            renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
            renderer.setSize(w, h);
            renderer.setPixelRatio(window.devicePixelRatio);
            webglContainer.appendChild(renderer.domElement);

            shaderMaterial = new THREE.ShaderMaterial({
                vertexShader, fragmentShader, transparent: true, depthWrite: false,
                uniforms: { uColorTex: { value: null }, uMouse: { value: new THREE.Vector2(-1, -1) }, uDispersion: { value: 0.0 }, uTime: { value: 0.0 } }
            });
            renderWebGL();
        }

        function resizeWebGL() {
            if(!camera || !renderer) return;
            const w = webglContainer.clientWidth || window.innerWidth;
            const h = webglContainer.clientHeight || window.innerHeight;
            if(w <= 0 || h <= 0) return;
            
            camera.aspect = w / h;
            camera.updateProjectionMatrix();
            renderer.setSize(w, h);
            if(currentItemData) updateWebGLMural(currentItemData.linePath, currentItemData.colorPath);
        }

        function updateWebGLMural(linePath, colorPath) {
            const textureLoader = new THREE.TextureLoader();
            textureLoader.setCrossOrigin('anonymous'); 
            const img = new Image(); 
            img.crossOrigin = 'anonymous';
            img.src = colorPath;
            
            img.onload = () => {
                const aspect = img.width / img.height;
                const planeHeight = 2.0; 
                const planeWidth = planeHeight * aspect;

                const fovRad = camera.fov * (Math.PI / 180);
                const distH = (planeHeight / 2) / Math.tan(fovRad / 2);
                const camAspect = camera.aspect || (window.innerWidth / window.innerHeight);
                const distW = (planeWidth / 2) / (Math.tan(fovRad / 2) * camAspect);
                
                defaultCameraZ = Math.max(distH, distW) * 1.05; 
                
                if (!isRevealed) {
                    camera.position.z = defaultCameraZ;
                    camera.position.x = 0;
                }
                
                cameraOffsetX = planeWidth * 0.25;

                if(lineMesh) { scene.remove(lineMesh); lineMesh.geometry.dispose(); lineMesh.material.dispose(); }
                if(colorPoints) { scene.remove(colorPoints); colorPoints.geometry.dispose(); }

                textureLoader.load(linePath, (lineTex) => {
                    lineMesh = new THREE.Mesh(new THREE.PlaneGeometry(planeWidth, planeHeight), new THREE.MeshBasicMaterial({ map: lineTex, transparent: true, depthWrite: false }));
                    lineMesh.position.z = -0.05; scene.add(lineMesh);
                });

                textureLoader.load(colorPath, (colorTex) => {
                    shaderMaterial.uniforms.uColorTex.value = colorTex;
                    const segmentsY = Math.floor(800 / aspect);
                    colorPoints = new THREE.Points(new THREE.PlaneGeometry(planeWidth, planeHeight, 800, segmentsY), shaderMaterial);
                    scene.add(colorPoints);
                });
            }

            img.onerror = () => {
                const canvas = document.createElement('canvas');
                canvas.width = 1024; canvas.height = 512;
                const ctx = canvas.getContext('2d');
                ctx.fillStyle = '#111'; ctx.fillRect(0, 0, 1024, 512);
                ctx.fillStyle = '#d4af37'; ctx.font = '40px sans-serif'; ctx.textAlign = 'center';
                ctx.fillText('图片加载失败 (可能受限于跨域)', 512, 256);
                const fallbackData = canvas.toDataURL();
                if (colorPath !== fallbackData) updateWebGLMural(fallbackData, fallbackData);
            }
        }

        const clock = new THREE.Clock();

            requestAnimationFrame(renderWebGL);
            if (currentTrack === 'cyber' && shaderMaterial) {
                shaderMaterial.uniforms.uTime.value = clock.getElapsedTime();
                renderer.render(scene, camera);
            }
        }

        function handleGestureLogic(landmarksList) {
            if(currentTrack !== 'cyber' || !shaderMaterial) return;

            const handsCount = landmarksList.length;
            const now = Date.now();
            const infoPanel = document.getElementById('floating-info');
            const infoText = document.getElementById('info-text');

            if (handsCount === 2) {
                pointerHistory = []; 
                const f1 = landmarksList[0][8];
                const f2 = landmarksList[1][8];
                const dist = Math.hypot(f1.x - f2.x, f1.y - f2.y);
                
                if (dist > 0.35 && !isRevealed) {
                    isRevealed = true;
                    gestureNeutralY = (f1.y + f2.y) / 2;

                    gsap.to(shaderMaterial.uniforms.uDispersion, { value: 1.0, duration: 1.5, ease: "power2.inOut" });
                    gsap.to(shaderMaterial.uniforms.uMouse.value, { x: -1, y: -1, duration: 0.5 });
                    gsap.to(camera.position, { x: cameraOffsetX, z: defaultCameraZ * 1.05, duration: 1.5, ease: "power2.inOut" });
                    infoPanel.classList.add('visible');
                    infoText.scrollTop = 0; 
                } 
                else if (dist < 0.15 && isRevealed) {
                    isRevealed = false;
                    gsap.to(shaderMaterial.uniforms.uDispersion, { value: 0.0, duration: 1.5, ease: "power2.inOut" });
                    gsap.to(camera.position, { x: 0, z: defaultCameraZ, duration: 1.5, ease: "power2.inOut" });
                    infoPanel.classList.remove('visible');
                }

            } else if (handsCount === 1 && !isRevealed) {
                const indexFinger = landmarksList[0][8];
                const targetX = 1.0 - indexFinger.x;
                const targetY = 1.0 - indexFinger.y;
                
                gsap.to(shaderMaterial.uniforms.uMouse.value, { x: targetX, y: targetY, duration: 0.2 });

                pointerHistory.push({ x: targetX, t: now });
                if(pointerHistory.length > 15) pointerHistory.shift(); 

                if (now - swipeCooldown > 1500 && pointerHistory.length > 5) {
                    const oldest = pointerHistory[0];
                    const newest = pointerHistory[pointerHistory.length - 1];
                    const dx = newest.x - oldest.x; 
                    const dt = newest.t - oldest.t;
                    
                    if (dt > 0 && Math.abs(dx) > 0.15 && (Math.abs(dx)/dt) > 0.0003) {
                        triggerSwipeFlash();
                        if (dx > 0) loadSeriesData(currentSeriesIndex + 1); 
                        else loadSeriesData(currentSeriesIndex - 1);        
                        swipeCooldown = now;
                        pointerHistory = [];
                    }
                }
            }

            if (isRevealed && handsCount > 0) {
                const maxScroll = infoText.scrollHeight - infoText.clientHeight;
                
                if (maxScroll > 0) {
                    let totalY = 0;
                    for (let i = 0; i < handsCount; i++) {
                        totalY += landmarksList[i][8].y;
                    }
                    const avgY = totalY / handsCount;

                    const deadzone = 0.08;
                    let scrollSpeed = 0;
                    
                    if (avgY < gestureNeutralY - deadzone) {
                        scrollSpeed = -((gestureNeutralY - deadzone) - avgY) * 150;
                    } 
                    else if (avgY > gestureNeutralY + deadzone) {
                        scrollSpeed = (avgY - (gestureNeutralY + deadzone)) * 150;
                    }

                    if (scrollSpeed !== 0) {
                        infoText.scrollTop += scrollSpeed;
                    }
                }
            }
        }

        // 【修复重构】将闪白光替换为电影级黑场过渡
        function triggerSwipeFlash() {
            const flash = document.getElementById('swipe-flash');
            // 瞬间切黑，完美遮挡 3D 贴图的切换
            flash.style.transition = 'none';
            flash.style.opacity = '1';
            
            // 延迟 100ms 后平滑淡出，仿佛人眼眨眼般的电影级转场
            setTimeout(() => { 
                flash.style.transition = 'opacity 1s ease-out'; 
                flash.style.opacity = '0'; 
            }, 100);
        }
